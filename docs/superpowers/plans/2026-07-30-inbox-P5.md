## §0.1 Decisions with real tradeoffs (CEO-grade, per Operating Discipline 12)

### D1 — Does a workspace change take effect live, or on restart?

`REPO_ROOT` is a module-load-time `const` in `guards.mjs`, and ~15 things in `server.mjs` derive from it at module scope (containment assertions, `engineWorktrees`, `targetIsGit`, both brokers, the seed warning, the report-back script paths).

- **Option A — hot-swap in-process.** Turn `REPO_ROOT` into a mutable accessor and rebuild every derived singleton. Cost: it mutates the security boundary (`applyAllowed` reads `REPO_ROOT`), creating a TOCTOU window where a request validated against root X applies against root Y; blast radius across all 487 runtime tests.
- **Option B — validate + persist, take effect on runtime restart.** `POST /runtime/workspace` validates and writes a 0600 config; `guards.mjs` reads it at load when the env var is unset; the response carries `restartRequired:true`. Electron already has the restart path (`ma:set-repo-root` → `restartSupervisor`, `main.cjs:182-188`).
- **Option C — env only (status quo).** Fails the scope.

**RECOMMENDATION: B.** It is the only option under which `applyAllowed`'s root is immutable for the lifetime of the process — which is precisely what makes the containment check trustworthy. Restart is a one-click experience in the packaged app and a one-line command in dev, and the UI states it plainly.
**Risk:** a persisted config on the dev machine could silently change `REPO_ROOT` for the existing test suite. **Mitigation:** Task 1 makes `repo-root.test.mjs`'s loader hermetic (points `RUNTIME_CONFIG_DIR` at a guaranteed-empty tmp dir) and Task 1 Step 6 greps the whole suite to prove no other test depends on the default root.

### D2 — Approve semantics for a worktree-backed Delivery

Today approve = `broker.applyChanges` writes the file into the **live repo working tree**. P5 changes it, for git-backed deliveries, to a **commit on the task branch** (rungs a/b) and leaves the operator's working tree untouched.

The program-plan §P5 exit wording says a remote-less repo gets its branch "merged". The build brief says "leave the branch pushed/local, mode='branch'". **We follow the brief:** an unattended merge into the checked-out branch is a materially bigger security decision than opening a PR, and there is no reviewer between the agent and `main` in that path. **DEVIATION DISCLOSED** — recorded here and in the delivery report; BOSS may reverse it at review.

### D3 — Migration v3 shape

v2 used a full table rebuild because SQLite cannot `ALTER` a `CHECK`. v3 only ADDS columns, and SQLite's `ALTER TABLE ... ADD COLUMN` permits a `CHECK` constraint. **Use three `ADD COLUMN`s, not a rebuild** — row preservation is then structural rather than something a copy statement has to get right, and idempotency is trivially provable.

### D4 — External PR link in Electron

`main.cjs:212` sets `setWindowOpenHandler(() => ({ action: "deny" }))` and `will-navigate` blocks non-loopback, so a `target=_blank` anchor is dead in the packaged app. Add one narrow IPC (`ma:open-external`) whose **main-process** side hard-allowlists `https:` + host `github.com` or `*.github.com`. In the browser the plain anchor already works.

// P5 — WORKSPACE CONFIG. "Point MissionAlpha at any git repo" needs exactly two
// things the runtime did not have: a VALIDATED choice, and somewhere durable and
// private to keep it. This module is both, and nothing else.
//
// SECURITY (sr-security, P5 is BLOCKING-security):
//   • Every check runs on the REALPATH. A symlink to $HOME is $HOME.
//   • Refused: non-existent · not a directory · not a git work tree · $HOME ·
//     the filesystem root · anything inside a runtime-owned worktree root ·
//     anything that would CONTAIN a runtime state dir (token/sandbox/worktree),
//     because server.mjs asserts those live OUTSIDE REPO_ROOT and would refuse
//     to boot — persisting such a path would brick the runtime.
//   • `git` is invoked as an argv array (no shell) with core.hooksPath
//     neutralised, so a hostile repo's hooks cannot run during validation.
//   • The config file is 0600 in a 0700 dir OUTSIDE the repo — same posture as
//     TOKEN_FILE (server.mjs L165) and mission.db (db.mjs L149).

import { execFileSync } from "node:child_process";
import { existsSync, statSync, realpathSync, mkdirSync, chmodSync, writeFileSync, readFileSync } from "node:fs";
import { resolve, sep, join } from "node:path";
import { tmpdir, homedir } from "node:os";

export const WORKSPACE_FILE_NAME = "workspace.json";

// Default config dir tracks the runtime's own out-of-repo state family so a test
// harness that isolates RUNTIME_TOKEN_DIR isolates this too.
export function defaultConfigDir(env = process.env) {
  if (env.RUNTIME_CONFIG_DIR) return resolve(env.RUNTIME_CONFIG_DIR);
  return join(env.RUNTIME_TOKEN_DIR ? resolve(env.RUNTIME_TOKEN_DIR) : join(tmpdir(), "missionalpha-runtime"), "config");
}

const isInside = (child, parent) => {
  const p = resolve(parent);
  const c = resolve(child);
  return c === p || c.startsWith(p + sep);
};

// The runtime state dirs that MUST stay outside the workspace (server.mjs L179-193).
function defaultStateDirs(env = process.env) {
  return [
    env.RUNTIME_SANDBOX_ROOT || join(tmpdir(), "missionalpha-sandboxes"),
    env.RUNTIME_TOKEN_DIR || join(tmpdir(), "missionalpha-runtime"),
    env.RUNTIME_WORKTREE_ROOT || join(tmpdir(), "missionalpha-worktrees"),
    ...(env.RUNTIME_EXEC_WT_ROOT ? [env.RUNTIME_EXEC_WT_ROOT] : []),
  ];
}
function defaultWorktreeRoots(env = process.env) {
  return [
    env.RUNTIME_WORKTREE_ROOT || join(tmpdir(), "missionalpha-worktrees"),
    ...(env.RUNTIME_EXEC_WT_ROOT ? [env.RUNTIME_EXEC_WT_ROOT] : []),
  ];
}

export function isGitWorkTree(dir, exec = execFileSync) {
  try {
    exec("git", ["-c", "core.hooksPath=/dev/null", "-C", dir, "rev-parse", "--git-dir"], {
      stdio: ["ignore", "pipe", "ignore"], timeout: 5000,
    });
    return true;
  } catch { return false; }
}

/**
 * Validate a candidate workspace. NEVER throws — returns a verdict so the HTTP
 * layer can 400 with a readable reason instead of 500-ing.
 * @returns {{ok:true, repoPath:string, isGit:true} | {ok:false, error:string}}
 */
export function validateWorkspace(input, opts = {}) {
  const env = opts.env || process.env;
  const exec = opts.exec || execFileSync;
  if (typeof input !== "string" || !input.trim()) return { ok: false, error: "repoPath required" };
  // eslint-disable-next-line no-control-regex
  if (/[\x00-\x1f]/.test(input)) return { ok: false, error: "repoPath contains control characters" };
  if (input.length > 4096) return { ok: false, error: "repoPath too long" };

  const raw = resolve(input.trim());
  if (!existsSync(raw)) return { ok: false, error: `path does not exist: ${raw}` };
  let repoPath;
  try { repoPath = realpathSync(raw); } catch (e) { return { ok: false, error: `cannot resolve path: ${e.message}` }; }
  let st;
  try { st = statSync(repoPath); } catch (e) { return { ok: false, error: `cannot stat path: ${e.message}` }; }
  if (!st.isDirectory()) return { ok: false, error: "path is not a directory" };

  if (repoPath === sep || resolve(repoPath) === resolve("/")) return { ok: false, error: "refusing the filesystem root" };
  let home;
  try { home = realpathSync(opts.homedir || homedir()); } catch { home = opts.homedir || homedir(); }
  if (repoPath === home) return { ok: false, error: "refusing your home directory — pick a project folder inside it" };

  for (const wt of (opts.worktreeRoots || defaultWorktreeRoots(env))) {
    if (existsSync(wt) && isInside(repoPath, realpathSync(wt))) {
      return { ok: false, error: "that folder is inside a runtime-owned worktree root — pick the source repo instead" };
    }
  }
  for (const sd of (opts.stateDirs || defaultStateDirs(env))) {
    const abs = resolve(sd);
    if (isInside(abs, repoPath)) {
      return { ok: false, error: `runtime state lives at ${abs}; it must stay OUTSIDE the workspace — pick a narrower folder` };
    }
  }
  if (!isGitWorkTree(repoPath, exec)) {
    return { ok: false, error: "not a git repo (run `git init` there first, or pick another folder)" };
  }
  return { ok: true, repoPath, isGit: true };
}

/** Persist a validated workspace. Refuses to write anything that fails validation. */
export function writeWorkspace(input, opts = {}) {
  const v = validateWorkspace(input, opts);
  if (!v.ok) return v;
  const dir = opts.configDir || defaultConfigDir(opts.env || process.env);
  mkdirSync(dir, { recursive: true, mode: 0o700 });
  try { chmodSync(dir, 0o700); } catch (e) { console.warn(`[workspace] chmod 0700 failed on ${dir}: ${e?.message}`); }
  const file = join(dir, WORKSPACE_FILE_NAME);
  writeFileSync(file, JSON.stringify({ repoPath: v.repoPath, savedAt: new Date().toISOString() }, null, 2), { mode: 0o600 });
  try { chmodSync(file, 0o600); } catch (e) { console.warn(`[workspace] chmod 0600 failed on ${file}: ${e?.message}`); }
  return { ok: true, repoPath: v.repoPath, file };
}

/**
 * The persisted workspace, or null. Deliberately CHEAP and total: a missing,
 * unreadable, corrupt, or vanished target yields null so the runtime falls back
 * to its default root rather than refusing to boot on a stale config.
 */
export function readWorkspace(opts = {}) {
  const dir = opts.configDir || defaultConfigDir(opts.env || process.env);
  try {
    const j = JSON.parse(readFileSync(join(dir, WORKSPACE_FILE_NAME), "utf-8"));
    const p = typeof j?.repoPath === "string" ? j.repoPath : null;
    if (!p || !existsSync(p)) return null;
    return realpathSync(p);
  } catch { return null; }
}

// P5 — APPROVE → COMMIT → (PR | BRANCH | APPLY).
//
// The Delivery engine already decides WHETHER work lands. This module decides HOW
// it leaves the runtime, and records which rung of the ladder actually executed:
//
//   (a) 'pr'     remote exists AND `gh auth status` succeeds
//                → git push -u origin <branch> → gh pr create --fill → prUrl
//   (b) 'branch' remote missing, gh absent/unauthenticated, or push/PR failed
//                → the commit stays on the local branch, honestly reported
//   (c) 'apply'  not a git repo / no worktree / nothing staged
//                → caller performs the pre-P5 direct broker apply
//
// NEVER FABRICATE SUCCESS: `mode` and `prUrl` are written only after the command
// that earns them returned ok. A failure demotes a rung; it never invents one.
//
// SECURITY (sr-security, P5 is BLOCKING-security):
//   • git and gh are argv arrays with shell:false. No interpolation anywhere.
//   • Every path is re-validated by the SAME applyAllowed the broker uses, then
//     passed after a literal `--` so a path can never be read as a flag.
//   • The staged set is verified against the approved set BEFORE committing; a
//     mismatch aborts rather than committing something nobody approved.
//   • core.hooksPath=/dev/null on every git call — a hostile repo's hooks never run.
//   • The gh token lives in gh's own keychain. This module never reads, stores,
//     logs or forwards a token, and strips *_TOKEN/*_KEY/*_SECRET from the child
//     env exactly like engineDispatch (server.mjs L1119-1123).
//   • Commit identity is a fixed, non-personal local identity passed per-invocation
//     with -c, so the operator's global git config is never read or written.

import { execFile } from "node:child_process";
import { existsSync } from "node:fs";
import { applyAllowed } from "./guards.mjs";

const EXEC_TIMEOUT_MS = Number(process.env.RUNTIME_PR_TIMEOUT_MS || 60_000);
const COMMIT_NAME = process.env.RUNTIME_COMMIT_NAME || "MissionAlpha";
const COMMIT_EMAIL = process.env.RUNTIME_COMMIT_EMAIL || "missionalpha@localhost";
const MAX_PATHS = 100;

// Child env WITHOUT credentials. gh reads its token from its own keychain/config,
// not from here, so stripping these costs nothing and removes an exfil surface.
function safeEnv() {

const CONVENTIONAL = { code: "feat", doc: "docs", post: "chore", command: "chore" };
const SUBJECT_CAP = 72;

/**
 * Commit message from the Delivery's own title/summary.
 * NO Claude / AI / "generated with" attribution — BOSS's global rule disables it.
 * The AGENT is credited as co-author because the agent is who wrote the diff.
 */
export function commitMessageFor({ title, summary, agentId, kind = "code" }) {
  const prefix = CONVENTIONAL[kind] || "chore";
  const raw = String(title || "change").replace(/\s+/g, " ").trim() || "change";
  const room = SUBJECT_CAP - prefix.length - 2;
  const subject = `${prefix}: ${raw.length > room ? `${raw.slice(0, room - 1)}…` : raw}`;
  const id = String(agentId || "agent").replace(/[^a-z0-9_-]/gi, "").slice(0, 64) || "agent";
  const body = String(summary || "").replace(/\r/g, "").trim().slice(0, 2000);
  return [subject, "", body, "", `Co-authored-by: ${id} <${id}@missionalpha.local>`]
    .filter((l, i, a) => !(l === "" && a[i - 1] === "")) // collapse blank runs
    .join("\n");
}

export function createPr(deps = {}) {
  const repoRoot = deps.repoRoot;
  const worktreeMgr = deps.worktreeMgr || null;
  const gh = deps.gh || defaultRunner();
  const audit = deps.audit || (() => {});
  const scrub = deps.scrub || ((s) => s);

  const git = (args, cwd) =>
    gh.run("git", ["-c", "core.hooksPath=/dev/null", ...args], { cwd, shell: false });

  async function isGit(dir) {
    if (!dir || !existsSync(dir)) return false;
    return (await git(["-C", dir, "rev-parse", "--git-dir"], dir)).ok;
  }

  async function hasRemote(dir) {
    const r = await git(["-C", dir, "remote", "get-url", "origin"], dir);
    return r.ok && r.stdout.trim().length > 0;
  }

  /** Honest capability probe for the Settings indicator. Never throws. */
  async function ghStatus() {
    const auth = await gh.run("gh", ["auth", "status"], { shell: false });
    const installed = auth.ok || !/ENOENT|not found|command not found/i.test(auth.stderr || "");
    return {
      installed: installed && !/ENOENT/i.test(auth.stderr || ""),
      authenticated: !!auth.ok,
      hasRemote: (await isGit(repoRoot)) ? await hasRemote(repoRoot) : false,
    };
  }

  /**
   * Land an approved Delivery. Returns
   * {mode:'pr'|'branch'|'apply', branch, prUrl, reason, refused[]}.
   * NEVER throws — the caller has already flipped the row to `approved`, so a
   * fault here must degrade the mode, not strand the decision.
   */
  async function deliverApproved({ taskId, paths, title, summary, agentId, kind }) {
    const out = { mode: "apply", branch: null, prUrl: null, reason: "", refused: [] };
    try {
      if (!worktreeMgr || !taskId) { out.reason = "no worktree manager or task key"; return out; }
      const wtPath = worktreeMgr.pathFor(taskId);
      if (!wtPath) { out.reason = "no worktree for this task"; return out; }
      if (!(await isGit(wtPath))) { out.reason = "worktree is not a git work tree"; return out; }

      // Re-run the SAME predicate the broker uses. A path the broker would refuse
      // must not reach a commit either — one rule, two exits.
      const approved = [];
      for (const p of (Array.isArray(paths) ? paths : []).slice(0, MAX_PATHS)) {
        if (applyAllowed(p)) approved.push(p);
        else out.refused.push(String(p).slice(0, 200));
      }
      if (!approved.length) { out.reason = "no allowed paths to commit"; return out; }

      const branch = worktreeMgr.branchOf(taskId);
      out.branch = branch;

      // `--` then pathspecs: an approved path can never be parsed as a flag.
      const add = await git(["-C", wtPath, "add", "--", ...approved], wtPath);
      if (!add.ok) { out.reason = `git add failed: ${scrub(add.stderr).slice(0, 300)}`; return out; }

      // Verify the staged set IS the approved set. Anything else aborts.
      const staged = await git(["-C", wtPath, "diff", "--cached", "--name-only"], wtPath);
      const stagedList = staged.stdout.split("\n").map((s) => s.trim()).filter(Boolean);
      if (!stagedList.length) { out.reason = "nothing to commit (no staged changes)"; return out; }
      const extra = stagedList.filter((p) => !approved.includes(p));
      if (extra.length) {
        await git(["-C", wtPath, "reset", "-q"], wtPath);
        out.reason = `staged set drifted from the approved set: ${extra.slice(0, 5).join(", ")}`;
        return out;
      }

      const msg = commitMessageFor({ title, summary, agentId, kind });
      const commit = await git(
        ["-C", wtPath, "-c", `user.name=${COMMIT_NAME}`, "-c", `user.email=${COMMIT_EMAIL}`, "commit", "-m", msg],
        wtPath,
      );
      if (!commit.ok) { out.reason = `git commit failed: ${scrub(commit.stderr).slice(0, 300)}`; return out; }
      out.mode = "branch"; // the commit exists — this is the floor from here on
      audit("delivery.committed", { taskId, branch, paths: approved });

      if (!(await hasRemote(wtPath))) { out.reason = "no remote configured — commit stays on the local branch"; return out; }
      const auth = await gh.run("gh", ["auth", "status"], { cwd: wtPath, shell: false });
      if (!auth.ok) { out.reason = `gh unavailable or unauthenticated — commit stays on the local branch`; return out; }

      const push = await git(["-C", wtPath, "push", "-u", "origin", branch], wtPath);
      if (!push.ok) { out.reason = `git push failed: ${scrub(push.stderr).slice(0, 300)}`; return out; }

      const pr = await gh.run("gh", ["pr", "create", "--fill", "--head", branch], { cwd: wtPath, shell: false });
      if (!pr.ok) { out.reason = `gh pr create failed: ${scrub(pr.stderr).slice(0, 300)}`; return out; }
      const url = (pr.stdout.match(/https:\/\/[^\s]+/) || [])[0] || null;
      if (!url) { out.reason = "gh pr create returned no URL"; return out; }

      out.mode = "pr";
      out.prUrl = url;
      out.reason = "";
      audit("delivery.pr", { taskId, branch, prUrl: url });
      return out;
    } catch (e) {
      out.reason = scrub(String(e?.message || e)).slice(0, 300);
      return out;
    }
  }

  return { deliverApproved, ghStatus, commitMessageFor };
}

async function authedJson<T>(path: string, init: RequestInit = {}): Promise<T> {
  const token = await runtimeToken();
