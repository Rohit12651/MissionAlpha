```js
// STOP button — active chat kill handles, keyed by agentId. Only ONE in-flight chat
// per agent (the inflight guard blocks concurrent), so a flat Map is enough. Set when
// the backend starts producing (cliReply spawn / mock loop), deleted when the turn
// settles or /stop fires. The map is the ONLY thing /stop can act on — a caller can
// never name a pid, only an agentId that must already own a registered handle.
/** @type {Map<string, () => void>} */
const chatKillMap = new Map();
// Agents whose current turn was stopped by the operator. Consumed by handleChat's
// save() so the persisted thread carries a visible "(interrupted)" marker.
/** @type {Set<string>} */
const interruptedChats = new Set();
// Grace period between SIGTERM and SIGKILL for a stopped `claude` child (BOSS ask:
// stream must be gone in under 2s, so the hard kill lands well inside that budget).
const STOP_SIGKILL_MS = 2000;

// Stop the in-flight chat turn for `agentId`. Returns true when a handle existed.
// Marks the thread interrupted so the saved history shows the turn was cut short.
function stopAgentChat(agentId) {
  const killFn = chatKillMap.get(agentId);
  if (!killFn) return false;
  interruptedChats.add(agentId);
  chatKillMap.delete(agentId);
  try { killFn(); } catch { /* child already gone */ }
  return true;
}
```

## Task 4: Wide skills roster → `src/data/skills.gen.json` (+ `source` on every entry)

**Design note (read before coding).** The generator currently scans only the repo-vendored
`.claude/` tree, which is why the catalog is reproducible in CI. P1 widens the scan to
`~/.claude/skills`, `~/.claude/plugins`, and the WaahyuTools repos named in `~/.claude/CLAUDE.md`,
and records WHERE each skill came from as `source`. Reproducibility is preserved two ways:
the emitted files are **committed artifacts** (regeneration is a manual `npm run gen:skills`,
exactly as today), and every root is **optional** — a machine without `~/.claude` falls back
to the vendored tree and still produces ≥1800 rows.

// Pure helpers for the P1 wide skills roster. No fs — unit-tested in
// scripts/lib/__tests__/skills-roster.test.mjs. The fs walk lives in skills-scan.mjs.

// A "repo roster" line in ~/.claude/CLAUDE.md is a single backtick span holding 5+
// comma-separated slug names. Anything shorter (an inline command like `npm run dev`)
// is not a roster and is ignored.
const MIN_ROSTER_ENTRIES = 5;
const SLUG_RE = /^[a-z0-9][a-z0-9._-]{1,63}$/;

/** Pure: extract the WaahyuTools repo names from a CLAUDE.md body. Sorted + deduped. */
export function parseToolRepos(claudeMd) {
  const found = new Set();
  for (const m of String(claudeMd || "").matchAll(/`([^`\n]+)`/g)) {
    const parts = m[1].split(",").map((s) => s.trim()).filter(Boolean);
    if (parts.length < MIN_ROSTER_ENTRIES) continue;
    if (!parts.every((p) => SLUG_RE.test(p))) continue;
    for (const p of parts) found.add(p);
  }
  return [...found].sort();
}

// Sources render in this order in the UI; unlisted prefixes fall between plugins and tools.
const SOURCE_RANK = (s) => (s === "curated" ? 0 : s === "user" ? 1 : s.startsWith("plugin:") ? 2 : s.startsWith("tools:") ? 4 : 3);

/** Pure: one row per skill id (first source wins), sorted source-rank → source → id. */
export function dedupeRoster(rows) {
  const seen = new Map();
  const ordered = [...rows].sort(
    (a, b) => SOURCE_RANK(a.source) - SOURCE_RANK(b.source) ||
      (a.source < b.source ? -1 : a.source > b.source ? 1 : 0) ||
      (a.id < b.id ? -1 : a.id > b.id ? 1 : 0),
  );
  for (const r of ordered) if (!seen.has(r.id)) seen.set(r.id, r);
  return [...seen.values()];
}

/** Pure: render the committed roster artifact. Deterministic; omits a redundant description. */
export function rosterJson(rows) {
  const out = rows.map((r) => (r.description && r.description !== r.id
    ? { id: r.id, name: r.name, source: r.source, family: r.family, description: r.description }
    : { id: r.id, name: r.name, source: r.source, family: r.family }));
  return JSON.stringify(out, null, 0) + "\n";
}

/** Pure: {source: count} summary for the generator's console line. */
export function countBySource(rows) {
  return rows.reduce((m, r) => ((m[r.source] = (m[r.source] || 0) + 1), m), {});
}

// fs walk for the P1 roster. Roots are passed in explicitly (never discovered inside the
// walker) so tests can point it at a fixture tree. Missing roots are skipped silently —
// a machine without ~/.claude or WaahyuTools must still generate a valid catalog.
import fs from "node:fs";
import path from "node:path";
import { familyOf, hintFrom } from "./skills-gen.mjs";

// Depth cap: real skill trees are ≤6 levels; the cap stops a pathological symlinked
// or generated tree from turning the scan into a hang.
const MAX_DEPTH = 8;

function walk(dir, baseRoot, source, out, seen, depth) {
  if (depth > MAX_DEPTH) return;
  let entries;
  try { entries = fs.readdirSync(dir, { withFileTypes: true }); } catch { return; }
  const skillFile = path.join(dir, "SKILL.md");
  if (fs.existsSync(skillFile)) {
    const rel = path.relative(baseRoot, dir).split(path.sep).join("/");
    const id = rel || path.basename(dir);
    const key = `${source} ${id}`;
    if (!seen.has(key)) {
      seen.add(key);
      let raw = "";
      try { raw = fs.readFileSync(skillFile, "utf8"); } catch { /* unreadable → id-only row */ }
      out.push({ id, name: id, source, family: familyOf(id, "skills"), description: hintFrom(raw, id) });
    }
  }
  for (const ent of entries.sort((a, b) => (a.name < b.name ? -1 : 1))) {
    // Skip hidden dirs (harness mirrors like .agents/.cursor duplicate every skill) and
    // never follow symlinks (loop-safe, escape-safe).
    if (!ent.isDirectory() || ent.name.startsWith(".") || ent.name === "node_modules") continue;
    const child = path.join(dir, ent.name);
    try { if (fs.lstatSync(child).isSymbolicLink()) continue; } catch { continue; }
    walk(child, baseRoot, source, out, seen, depth + 1);
  }
}

/** Scan a list of {dir, source} roots. Deterministic; missing dirs are skipped. */
export function scanRoots(roots) {
  const out = [];
  const seen = new Set();
  for (const { dir, source } of roots) {
    if (!dir || !fs.existsSync(dir)) continue;
    walk(dir, dir, source, out, seen, 0);
  }
  return out;
}

/** Build the root list for a machine. Every root is optional. */
export function discoverRoots({ homeDir, toolsDir, repoDir, toolRepos = [] }) {
  const roots = [];
  // 1. In-repo curated skills win on id collision (scanned first).
  roots.push({ dir: path.join(repoDir, ".claude", "skills-curated"), source: "curated" });
  // 2. The operator's own skill tree.
  roots.push({ dir: path.join(homeDir, ".claude", "skills"), source: "user" });
  // 3. Installed plugins — each top-level plugin dir becomes its own source.
  const pluginsDir = path.join(homeDir, ".claude", "plugins");
  for (const name of listDirs(pluginsDir)) roots.push({ dir: path.join(pluginsDir, name), source: `plugin:${name}` });
  // 4. WaahyuTools skill repos named in ~/.claude/CLAUDE.md.
  for (const repo of toolRepos) roots.push({ dir: path.join(toolsDir, repo), source: `tools:${repo}` });
  // 5. Fallback: the repo-vendored tree keeps CI reproducible when the roots above are absent.
  roots.push({ dir: path.join(repoDir, ".claude", "skills"), source: "user" });
  return roots;
}

function listDirs(dir) {
  try {
    return fs.readdirSync(dir, { withFileTypes: true })
      .filter((e) => e.isDirectory() && !e.name.startsWith("."))
      .map((e) => e.name).sort();
  } catch { return []; }
}

  keyOf: (s: MissionSkill) => string = (s) => s.family,

```ts
// P1-F6 — the CEO's team proposal. The CEO may end a boss-thread reply with a fenced
// ```team block; each line is `<agent-id> <role title> | <skill,skill>`. This module is
// PURE (no fetch, no DOM) so the whole proposal path is unit-testable, and it mirrors
// parseDispatch's shape: {clean, members}.
//
// SECURITY: this parser is a FILTER, not a trust boundary. Every field is re-validated by
// the runtime's factory.validateSpec on POST /factory/create (id regex, reserved built-in
// ids, skill ⊆ catalog, persona length + secret-shape). Nothing here can widen capability:
// the skill list is checked against the catalog the server itself served us, and
// `integrations` is always empty, so no minted agent can post anywhere.

/** One proposed team member. */
export interface TeamMember {
  id: string;
  role: string;
  skills: string[];
}

export interface TeamParse {
  clean: string;
  members: TeamMember[];
}

/** Mirrors the factory's ID_RE (`^[a-z][a-z0-9-]{2,48}$`). */
const ID_RE = /^[a-z][a-z0-9-]{2,48}$/;
const ROLE_MAX = 80;
/** Hard cap on a single proposal so one reply can't queue 50 agent creations. */
export const TEAM_MAX = 8;

/**
 * Extract a ```team block. `knownSkills` is the governed factory catalog (from
 * GET /factory/skills) — a line naming any skill outside it is dropped whole, so the
 * confirm card can never show an agent the server would then reject.
 */
export function parseTeamBlock(content: string, knownSkills: ReadonlySet<string>): TeamParse {
  const m = /```team\s*\n([\s\S]*?)```/.exec(content);
  if (!m) return { clean: content, members: [] };

  const members: TeamMember[] = [];
  const seen = new Set<string>();
  for (const raw of m[1].split("\n")) {
    if (members.length >= TEAM_MAX) break;
    const line = raw.trim();
    if (!line) continue;

    const [left, right] = line.split("|");
    const head = left.trim().split(/\s+/);
    const id = (head.shift() || "").toLowerCase();
    const role = head.join(" ").slice(0, ROLE_MAX).trim();
    if (!ID_RE.test(id) || !role || seen.has(id)) continue;

    const skills = (right ?? "")
      .split(",")
      .map((s) => s.trim().toLowerCase())
      .filter(Boolean);
    if (skills.some((s) => !knownSkills.has(s))) continue; // unknown skill → drop the member

    seen.add(id);
    members.push({ id, role, skills: [...new Set(skills)] });
  }
  return { clean: content.replace(m[0], "").trim(), members };
}

/** Build the POST /factory/create payload for one member. The server re-validates it all. */
export function teamMemberToSpec(member: TeamMember, job: string) {
  const brief = job.replace(/\s+/g, " ").trim().slice(0, 600);
  const persona =
    `You are the ${member.role} on a team the CEO assembled for this job: ${brief || "an unspecified job"}. ` +
    `Own your lane end to end, report progress to the CEO in /caveman full, and flag blockers early. ` +
    `You never post externally and never claim work you did not do.`;
  return {
    id: member.id,
    role: member.role,
    title: member.role,
    persona: persona.slice(0, 6000),
    skills: member.skills,
    tools: [] as string[],   // derived server-side from the chosen skills
    integrations: [] as string[], // never granted from a chat proposal
  };
}
```
